# -----------------------------------------------------------------------------
# Prebuilt dependency superbuild for RadarSimCpp
# -----------------------------------------------------------------------------
# This project contains no source code of its own. For every third-party
# dependency that RadarSimCpp consumes as a prebuilt static library it
# downloads a pinned release, builds it against a shared ABI baseline, and
# stages the headers and libraries into ./output in the layout the consumer
# expects:
#
#   output/<dep>/include/              - portable headers
#   output/<dep>/include_<platform>/   - headers generated by the build
#   output/<dep>/lib_<platform>/       - static libraries (.a / .lib)
#
# The dependencies currently covered:
#
#   hdf5      - data I/O, always needed
#   mbedtls   - license verification, needed when radarsimcpp is configured
#               with -DENABLE_LICENSE=ON
#
# Each dependency is built by its own CMake project under recipes/, driven from
# here with ExternalProject_Add. That isolation is not cosmetic: HDF5 and
# mbedTLS both set global cache variables (BUILD_TESTING, BUILD_SHARED_LIBS,
# ENABLE_PROGRAMS, ...) and both define targets with generic names, so building
# them in one CMake cache makes them fight each other. A separate cache per
# recipe means adding the next dependency cannot disturb the existing ones.
#
# Normal usage is through ./build.sh (Linux/macOS) or build.bat (Windows).
# -----------------------------------------------------------------------------

# HDF5 2.x requires CMake 3.26 or newer.
cmake_minimum_required(VERSION 3.26)

project(radarsimcpp_deps LANGUAGES NONE)

include(ExternalProject)

list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake")
include(platform)

# -----------------------------------------------------------------------------
# Options
# -----------------------------------------------------------------------------
option(DEPS_BUILD_HDF5 "Build and stage HDF5" ON)
option(DEPS_BUILD_MBEDTLS "Build and stage mbedTLS" ON)

set(DEPS_OUTPUT_DIR "${PROJECT_SOURCE_DIR}/output"
    CACHE PATH "Directory the staged headers and libraries are written to")

# Version pins. Neither of these can be tracked by Dependabot (one is a Git tag
# consumed by FetchContent, the other a release tarball URL), so bumping them is
# a deliberate one-line edit followed by a CI run.
set(HDF5_GIT_TAG "2.1.1" CACHE STRING "HDF5 release tag to build")
set(MBEDTLS_VERSION "4.2.0" CACHE STRING "mbedTLS release version to build")

set(DEPS_OSX_DEPLOYMENT_TARGET "" CACHE STRING
    "macOS deployment target to build the prebuilt libraries against (empty = toolchain default)")

message(STATUS "Staging platform: ${DEPS_PLATFORM}")
message(STATUS "Output directory: ${DEPS_OUTPUT_DIR}")

# -----------------------------------------------------------------------------
# Arguments forwarded to every recipe
# -----------------------------------------------------------------------------
# The recipes are separate CMake caches, so anything that has to match across
# dependencies must be passed explicitly.
set(DEPS_COMMON_CMAKE_ARGS
    "-DCMAKE_BUILD_TYPE:STRING=${CMAKE_BUILD_TYPE}"
    "-DCMAKE_C_COMPILER:STRING=${CMAKE_C_COMPILER}"
    "-DCMAKE_CXX_COMPILER:STRING=${CMAKE_CXX_COMPILER}"
    "-DDEPS_OUTPUT_DIR:PATH=${DEPS_OUTPUT_DIR}"
    "-DDEPS_OSX_DEPLOYMENT_TARGET:STRING=${DEPS_OSX_DEPLOYMENT_TARGET}"
)

# Passed via CMAKE_CACHE_ARGS instead so the embedded semicolons survive.
set(DEPS_COMMON_CACHE_ARGS
    "-DCMAKE_OSX_ARCHITECTURES:STRING=${CMAKE_OSX_ARCHITECTURES}"
)

# -----------------------------------------------------------------------------
# add_dep_recipe(<name>) - build recipes/<name> in its own CMake cache
# -----------------------------------------------------------------------------
# BUILD_ALWAYS is on because a recipe's real work (fetching, building, staging)
# is invisible to the superbuild's dependency graph; without it a second
# `cmake --build` would decide there is nothing to do and silently skip the
# staging step.
function(add_dep_recipe name)
    ExternalProject_Add(${name}
        SOURCE_DIR "${PROJECT_SOURCE_DIR}/recipes/${name}"
        BINARY_DIR "${PROJECT_BINARY_DIR}/${name}"
        CMAKE_ARGS ${DEPS_COMMON_CMAKE_ARGS} ${ARGN}
        CMAKE_CACHE_ARGS ${DEPS_COMMON_CACHE_ARGS}
        BUILD_COMMAND ${CMAKE_COMMAND} --build <BINARY_DIR> --config $<CONFIG> --parallel
        BUILD_ALWAYS TRUE
        INSTALL_COMMAND ""
        USES_TERMINAL_DOWNLOAD TRUE
        USES_TERMINAL_CONFIGURE TRUE
        USES_TERMINAL_BUILD TRUE
    )
endfunction()

set(DEPS_ENABLED "")

if(DEPS_BUILD_HDF5)
    message(STATUS "Enabled dependency: hdf5 @ ${HDF5_GIT_TAG}")
    add_dep_recipe(hdf5 "-DHDF5_GIT_TAG:STRING=${HDF5_GIT_TAG}")
    list(APPEND DEPS_ENABLED hdf5)
    set(DEPS_VERSION_hdf5 "${HDF5_GIT_TAG}")
endif()

if(DEPS_BUILD_MBEDTLS)
    message(STATUS "Enabled dependency: mbedtls @ ${MBEDTLS_VERSION}")
    add_dep_recipe(mbedtls "-DMBEDTLS_VERSION:STRING=${MBEDTLS_VERSION}")
    list(APPEND DEPS_ENABLED mbedtls)
    set(DEPS_VERSION_mbedtls "${MBEDTLS_VERSION}")
endif()

if(NOT DEPS_ENABLED)
    message(FATAL_ERROR "No dependencies selected. Enable at least one of DEPS_BUILD_HDF5 / DEPS_BUILD_MBEDTLS.")
endif()

# -----------------------------------------------------------------------------
# Manifest
# -----------------------------------------------------------------------------
# Records what this build produced, so CI and anyone reading a downloaded
# archive can tell which version and platform they are holding without having
# to re-derive it from the CMake cache.
set(DEPS_MANIFEST "")
foreach(_dep IN LISTS DEPS_ENABLED)
    string(APPEND DEPS_MANIFEST "${_dep} ${DEPS_VERSION_${_dep}} ${DEPS_PLATFORM}\n")
endforeach()

file(MAKE_DIRECTORY "${DEPS_OUTPUT_DIR}")
file(WRITE "${DEPS_OUTPUT_DIR}/manifest.txt" "${DEPS_MANIFEST}")

# -----------------------------------------------------------------------------
# Release packaging
# -----------------------------------------------------------------------------
# `cmake --build build --target package-release` turns the staged output into
# one archive per dependency plus a SHA-256 checksum, ready to attach to a
# GitHub release. The consumer pins those checksums, so this target is the
# producer half of the download path in radarsimcpp's PrebuiltDeps.cmake.
set(DEPS_DIST_DIR "${PROJECT_SOURCE_DIR}/dist"
    CACHE PATH "Directory release archives are written to")

add_custom_target(package-release
    COMMENT "Packaging staged dependencies into ${DEPS_DIST_DIR}"
)

foreach(_dep IN LISTS DEPS_ENABLED)
    add_custom_command(TARGET package-release POST_BUILD
        COMMAND ${CMAKE_COMMAND}
            "-DDEP=${_dep}"
            "-DVERSION=${DEPS_VERSION_${_dep}}"
            "-DPLATFORM=${DEPS_PLATFORM}"
            "-DSTAGE_DIR=${DEPS_OUTPUT_DIR}/${_dep}"
            "-DDIST_DIR=${DEPS_DIST_DIR}"
            -P "${PROJECT_SOURCE_DIR}/cmake/package_release.cmake"
        VERBATIM
    )
endforeach()

add_dependencies(package-release ${DEPS_ENABLED})
